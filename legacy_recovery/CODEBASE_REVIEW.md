### 2.2 Common mathematical operations

- The clean target is the same resized Shepp-Logan phantom for every sample. A noisy input is `clip(clean + Gaussian noise, 0, 1)`. Thus the noise is Gaussian **before clipping**; the effective input error after clipping is not an unrestricted Gaussian variable, particularly near 0 and 1.
- Stage 3 uses image shape `(512, 512, 1)` and batch shape `(1, 512, 512, 1)`. The final axis is one grayscale channel; the leading axis is the batch.
- MSE averages squared pixel differences. The training loss `"mse"` and separately named `MeanSquaredError(name="mse")` measure the same squared-error quantity, but occupy distinct history/result keys.
- `tf.image.psnr(..., max_val=1.0)` computes per-image PSNR. The custom functions return its batch mean. With batch size 1, each batch contributes one image; the reported PSNR is an average of image PSNR values, not generally `10*log10(1 / average_image_MSE)`.
- `model.compile` connects the architecture, optimizer, loss, and reporting metrics. `model.fit` performs forward passes, loss calculation, gradient-based weight updates, and validation passes. `model.evaluate` measures metrics without training. `model.predict` or `model(noisy, training=False)` produces recovered images.
- Training sets shuffle index order. Stateless noise is tied to the index, so an index receives the same noisy image on each visit; there is no fresh noise realization every epoch. Validation/test sets are not shuffled.

```python
import os
import numpy as np
import matplotlib.pyplot as plt

from skimage.data import shepp_logan_phantom
from skimage.transform import resize


# =========================================================
# STAGE ONE - PART A
# GENERATE AND DEGRADE THE SHEPP-LOGAN PHANTOM
# =========================================================

# Generate the original Shepp-Logan phantom
phantom = shepp_logan_phantom()

print("Type:", type(phantom))
print("Original shape:", phantom.shape)


# Resize the phantom to 512 x 512
phantom = resize(
    phantom,
    (512, 512),
    anti_aliasing=True
).astype(np.float32)

print("New shape:", phantom.shape)
print("Minimum pixel value:", phantom.min())
print("Maximum pixel value:", phantom.max())


# Gaussian noise parameters
mean = 0
sigma = 0.1


# Create a reproducible random-number generator
rng = np.random.default_rng(42)


# Generate Gaussian noise
noise = rng.normal(
    loc=mean,
    scale=sigma,
    size=phantom.shape
).astype(np.float32)


# Add the noise to the clean phantom
noisy_phantom = phantom + noise


# Keep all pixel values between 0 and 1
noisy_phantom = np.clip(
    noisy_phantom,
    0,
    1
)


# Create a folder where results will be saved
os.makedirs("results", exist_ok=True)


# Display clean and noisy phantom
plt.figure(figsize=(10, 5))

plt.subplot(1, 2, 1)
plt.imshow(phantom, cmap="gray")
plt.title("Clean Shepp-Logan Phantom")
plt.axis("off")

plt.subplot(1, 2, 2)
plt.imshow(noisy_phantom, cmap="gray")
plt.title("Noisy Shepp-Logan Phantom")
plt.axis("off")

plt.tight_layout()

# Save the result
plt.savefig(
    "results/clean_and_noisy_phantom.png",
    dpi=300,
    bbox_inches="tight"
)

plt.show()


# =========================================================
# STAGE ONE - PART B
# CREATE MULTIPLE NOISY SAMPLES
# =========================================================

num_samples = 500


# Create arrays to store the images
noisy_images = np.empty(
    (num_samples, 512, 512, 1),
    dtype=np.float32
)

clean_images = np.empty(
    (num_samples, 512, 512, 1),
    dtype=np.float32
)


# Generate 500 different noisy versions of the same phantom
for i in range(num_samples):

    noise = rng.normal(
        loc=mean,
        scale=sigma,
        size=phantom.shape
    ).astype(np.float32)

    noisy = phantom + noise

    noisy = np.clip(
        noisy,
        0,
        1
    )

    # Store noisy image
    noisy_images[i, :, :, 0] = noisy

    # Store corresponding clean target
    clean_images[i, :, :, 0] = phantom


print("\nDataset created successfully")
print("Noisy dataset shape:", noisy_images.shape)
print("Clean dataset shape:", clean_images.shape)


# =========================================================
# SPLIT DATA INTO TRAINING, VALIDATION AND TESTING
# =========================================================

# Shuffle the positions of the 500 samples
indices = rng.permutation(num_samples)


# 70% training = 350 images
train_indices = indices[:350]

# 15% validation = 75 images
val_indices = indices[350:425]

# 15% testing = 75 images
test_indices = indices[425:500]


print("\nTraining samples:", len(train_indices))
print("Validation samples:", len(val_indices))
print("Testing samples:", len(test_indices))


# =========================================================
# DISPLAY ONE TRAINING EXAMPLE
# =========================================================

sample_index = train_indices[0]

plt.figure(figsize=(10, 5))


# Noisy input
plt.subplot(1, 2, 1)

plt.imshow(
    noisy_images[sample_index, :, :, 0],
    cmap="gray"
)

plt.title("Training Input - Noisy Phantom")
plt.axis("off")


# Clean target
plt.subplot(1, 2, 2)

plt.imshow(
    clean_images[sample_index, :, :, 0],
    cmap="gray"
)

plt.title("Training Target - Clean Phantom")
plt.axis("off")


plt.tight_layout()


# Save the training example
plt.savefig(
    "results/stage1_training_pair.png",
    dpi=300,
    bbox_inches="tight"
)

plt.show()


print("\nStage One completed successfully.")
```

```python
    Concatenate
)


def build_unet(input_shape=(512, 512, 1)):

    inputs = Input(shape=input_shape)

    # Encoder level 1
    c1 = Conv2D(8, (3, 3), activation="relu", padding="same")(inputs)
    c1 = Conv2D(8, (3, 3), activation="relu", padding="same")(c1)
    p1 = MaxPooling2D((2, 2))(c1)

    # Encoder level 2
    c2 = Conv2D(16, (3, 3), activation="relu", padding="same")(p1)
    c2 = Conv2D(16, (3, 3), activation="relu", padding="same")(c2)
    p2 = MaxPooling2D((2, 2))(c2)

    # Encoder level 3
    c3 = Conv2D(32, (3, 3), activation="relu", padding="same")(p2)
    c3 = Conv2D(32, (3, 3), activation="relu", padding="same")(c3)
    p3 = MaxPooling2D((2, 2))(c3)

    # Bottleneck
    b = Conv2D(64, (3, 3), activation="relu", padding="same")(p3)
    b = Conv2D(64, (3, 3), activation="relu", padding="same")(b)

    # Decoder level 3
    u3 = Conv2DTranspose(
        32,
        (2, 2),
        strides=(2, 2),
        padding="same"
    )(b)

    u3 = Concatenate()([u3, c3])

    c4 = Conv2D(32, (3, 3), activation="relu", padding="same")(u3)
    c4 = Conv2D(32, (3, 3), activation="relu", padding="same")(c4)

    # Decoder level 2
    u2 = Conv2DTranspose(
        16,
        (2, 2),
        strides=(2, 2),
        padding="same"
    )(c4)

    u2 = Concatenate()([u2, c2])

    c5 = Conv2D(16, (3, 3), activation="relu", padding="same")(u2)
    c5 = Conv2D(16, (3, 3), activation="relu", padding="same")(c5)

    # Decoder level 1
    u1 = Conv2DTranspose(
        8,
        (2, 2),
        strides=(2, 2),
        padding="same"
    )(c5)

    u1 = Concatenate()([u1, c1])

    c6 = Conv2D(8, (3, 3), activation="relu", padding="same")(u1)
    c6 = Conv2D(8, (3, 3), activation="relu", padding="same")(c6)

    # Output
    outputs = Conv2D(
        1,
        (1, 1),
        activation="sigmoid"
    )(c6)

    model = Model(
        inputs=inputs,
        outputs=outputs,
        name="Simplified_UNet"
    )

    return model

if __name__ == "__main__":
    model = build_unet()
    model.summary()
```

```python
import os
import numpy as np
import tensorflow as tf

from skimage.data import shepp_logan_phantom
from skimage.transform import resize

from stage2_unet import build_unet


SEED = 42
IMAGE_SIZE = 512
NUM_SAMPLES = 500
TEST_SAMPLES = 75

RESULTS_DIR = "results/stage3_cross_validation"


# ------------------------------------------------------------
# CLEAN PHANTOM
# ------------------------------------------------------------

phantom = shepp_logan_phantom()

phantom = resize(
    phantom,
    (IMAGE_SIZE, IMAGE_SIZE),
    anti_aliasing=True
).astype(np.float32)

phantom = phantom[..., np.newaxis]

clean_image = tf.constant(
    phantom,
    dtype=tf.float32
)


# ------------------------------------------------------------
# RECREATE EXACT SAME TEST SPLIT
# ------------------------------------------------------------

all_indices = np.arange(
    NUM_SAMPLES,
    dtype=np.int32
)

rng = np.random.default_rng(SEED)

shuffled_indices = rng.permutation(
    all_indices
)

test_indices = shuffled_indices[
    :TEST_SAMPLES
]


# ------------------------------------------------------------
# RECREATE EXACT SAME NOISY IMAGES
# ------------------------------------------------------------

def create_noisy_pair(index):

    index = tf.cast(index, tf.int32)

    sample_seed = tf.stack([
        tf.constant(SEED, dtype=tf.int32),
        index
    ])

    noise = tf.random.stateless_normal(
        shape=tf.shape(clean_image),
        seed=sample_seed,
        mean=0.0,
        stddev=0.1,
        dtype=tf.float32
    )

    noisy = clean_image + noise

    noisy = tf.clip_by_value(
        noisy,
        0.0,
        1.0
    )

    return noisy, clean_image


test_dataset = (
    tf.data.Dataset
    .from_tensor_slices(test_indices)
    .map(create_noisy_pair)
    .batch(1)
)


# ------------------------------------------------------------
# BASELINE SSIM
# ------------------------------------------------------------

baseline_ssim = []


for noisy, clean in test_dataset:

    value = tf.image.ssim(
        clean,
        noisy,
        max_val=1.0
    )

    baseline_ssim.append(
        float(value.numpy()[0])
    )


print()
print(
    "Noisy baseline SSIM:",
    np.mean(baseline_ssim)
)


# ------------------------------------------------------------
# OPTIMIZER MODELS
# ------------------------------------------------------------

optimizers = [
    "sgd",
    "rmsprop",
    "adam"
]


for optimizer_name in optimizers:

    tf.keras.backend.clear_session()

    model = build_unet(

    )

        RESULTS_DIR,
        f"{optimizer_name}_final.weights.h5"
    )

    model.load_weights(
        weights_path
    )

    ssim_values = []


    for noisy, clean in test_dataset:

        prediction = model(
            noisy,
            training=False
        )

        value = tf.image.ssim(

            prediction,

        )

        ssim_values.append(
            float(value.numpy()[0])

    mean_ssim = np.mean(

print("SSIM CALCULATION COMPLETED.")
```

```python
import numpy as np
import tensorflow as tf

from skimage.data import shepp_logan_phantom
from skimage.transform import resize

from stage2_unet import build_unet


SEED = 42
IMAGE_SIZE = 512
TRAIN_SAMPLES = 340
VAL_SAMPLES = 85
EPOCHS = 10
BATCH_SIZE = 1

MEAN = 0.0
SIGMA = 0.1


tf.keras.utils.set_random_seed(SEED)


phantom = shepp_logan_phantom()

phantom = resize(
    phantom,
    (IMAGE_SIZE, IMAGE_SIZE),
    anti_aliasing=True
).astype(np.float32)

phantom = phantom[..., np.newaxis]

clean_image = tf.constant(
    phantom,
    dtype=tf.float32
)


def create_noisy_pair(index):

    index = tf.cast(index, tf.int32)

    seed = tf.stack([
        tf.constant(SEED, dtype=tf.int32),
        index
    ])

    noise = tf.random.stateless_normal(
        tf.shape(clean_image),
        seed=seed,
        mean=MEAN,
        stddev=SIGMA,
        dtype=tf.float32
    )

    noisy = tf.clip_by_value(
        clean_image + noise,
        0.0,
        1.0
    )

    return noisy, clean_image


def make_dataset(indices, training=False):

    ds = tf.data.Dataset.from_tensor_slices(indices)

    if training:
        ds = ds.shuffle(
            len(indices),
            seed=SEED
        )

    ds = ds.map(
        create_noisy_pair,
        num_parallel_calls=1
    )

    ds = ds.batch(BATCH_SIZE)

    ds = ds.prefetch(1)

    return ds


train_indices = np.arange(
    0,
    TRAIN_SAMPLES,
    dtype=np.int32
)

val_indices = np.arange(
    TRAIN_SAMPLES,
    TRAIN_SAMPLES + VAL_SAMPLES,
    dtype=np.int32
)


train_ds = make_dataset(
    train_indices,
    training=True
)

val_ds = make_dataset(
    val_indices,
    training=False
)


def psnr_metric(y_true, y_pred):

    return tf.reduce_mean(
        tf.image.psnr(
            y_true,
            y_pred,
            max_val=1.0
        )
    )


model = build_unet(
    input_shape=(512, 512, 1)
)


# IMPORTANT:
# lower learning rate than previous Adam run
optimizer = tf.keras.optimizers.Adam(
    learning_rate=0.0001
)


model.compile(
    optimizer=optimizer,
    loss="mse",
    metrics=[
        tf.keras.metrics.MeanSquaredError(
            name="mse"
        ),
        psnr_metric
    ]
)


history = model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=EPOCHS,
    verbose=1
)


results = model.evaluate(
    val_ds,
    verbose=0,
    return_dict=True
)


print()
print("ADAM DIAGNOSTIC RESULTS")
print("-----------------------")

print(
    "Validation MSE:",
    results["mse"]
)

print(
    "Validation PSNR:",
    results["psnr_metric"]
)


for noisy_batch, clean_batch in val_ds.take(1):

    prediction = model.predict(
        noisy_batch,
        verbose=0
    )


prediction = prediction[0]

print()
print("Prediction statistics:")
print("Minimum:", prediction.min())
print("Maximum:", prediction.max())
print("Mean:", prediction.mean())

print()
print("Clean image statistics:")
print("Minimum:", phantom.min())
print("Maximum:", phantom.max())
print("Mean:", phantom.mean())
```
